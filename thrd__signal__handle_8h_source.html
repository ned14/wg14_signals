42 // MSVC may be missing necessary signal support
43 typedef uint32_t sigset_t;
44 static inline void sigemptyset(sigset_t *ss)
45 {
46 *ss = 0;
47 }
48 static inline void sigfillset(sigset_t *ss)
49 {
50 *ss = UINT32_MAX;
51 }
52 // The shifts below are bounds-checked against the 32-signal bit set
53 // (analysis.md 4.5): for signo outside [1, 32], 1u << (signo - 1) is
54 // undefined behaviour. sigaddset/sigdelset become no-ops out of range;
55 // sigismember is kept total (returns false) so the Windows sigfillset_*
56 // lazy-init checks never read a torn set (plans/ideas.md 4.3). These
57 // helpers run in signal-handler context too, so no error reporting.
58 static inline void sigaddset(sigset_t *ss, const int signo)
59 {
60 if(signo >= 1 && signo <= 32)
61 {
62 *ss |= (1u << (signo - 1));
63 }
64 }
65 static inline void sigdelset(sigset_t *ss, const int signo)
66 {
67 if(signo >= 1 && signo <= 32)
68 {
69 *ss &= ~(1u << (signo - 1));
70 }
71 }
72 static inline bool sigismember(const sigset_t *ss, const int signo)
73 {
74 return (signo >= 1 && signo <= 32) && (*ss & (1u << (signo - 1))) != 0;
75 }
76

145 // The arg counting above depends on __VA_OPT__ (C23/C++20, supported as a
146 // GNU/Clang/MSVC extension in older modes). Verify at compile time that the
147 // counting machinery returns the right counts: on a compiler without
148 // __VA_OPT__ the expansion below is a hard preprocessing error, and on a
149 // compiler whose counting is broken the assertion fails — either way the
150 // defect surfaces instead of silently mis-dispatching. (The zero-argument
151 // path — the comma-suppression case __VA_OPT__ exists for — is exercised by
152 // test/sigfence_fence_test.c, whose TU suppresses the -Wpedantic diagnostic
153 // that an empty variadic-macro call triggers; a public header must not.
154 // The counts asserted here — 3 and 5 — are in the 1..8 range that the
155 // non-__VA_OPT__ fallback counting also handles, so the assert is valid on
156 // every supported compiler, including MSVC C++14/17.)
157 WG14_SIGNALS_STATIC_ASSERT(
158 ((WG14_SIGNALS_SIGFENCE_COUNT_ARGS_MAX8(a, b, c) == 3) &&
159 (WG14_SIGNALS_SIGFENCE_COUNT_ARGS_MAX8(a, b, c, d, e) == 5)),
160 "wg14_signals: sigfence() argument counting is broken (requires __VA_OPT__ "
161 "on this compiler, or a counting-broken preprocessor)");

203 // Compilers without extended inline asm (e.g. MSVC), or with it disabled via
204 // -DDISABLE_INLINE_ASM (Fil-C): force the listed local variables to be
205 // memory-resident, and their values reloaded afterwards, as the "+m" operands
206 // and "memory" clobber above do.
207 // WG14_SIGNALS_SIGFENCE_ESCAPE() (1) stores each variable's address into a
208 // volatile sink so the address escapes to observable memory, and (2) performs
209 // a volatile read of one byte of the object -- char may alias any object
210 // (C11 6.5p7), and volatile accesses are observable behaviour (C11 5.1.2.3),
211 // so no optimizer, link-time code generation (/GL /LTCG) included, may
212 // eliminate or reorder them: the value is committed to memory before the
213 // fence and must be reloaded after it. No out-of-line function is needed, so
214 // the fence cannot be defeated by the optimizer inlining a helper away, and
215 // the sink is per-TU static, so header-only consumers need no library
216 // symbols.
